BULK_THRESHOLD = 10
BULK_DISCOUNT = 0.9

def line_cost(unit, quantity):
    amount = unit * quantity
    if quantity >= BULK_THRESHOLD:
        return amount * BULK_DISCOUNT
    return amount

def order_total(lines):
    total = 0
    for unit, quantity in lines:
        total = total + line_cost(unit, quantity)
    return total

def cheapest(lines):
    best = None
    for unit, quantity in lines:
        if best is None or unit < best:
            best = unit
    return best

def positions_of(target, values):
    found = []
    for i in range(len(values)):
        if values[i] == target:
            found.append(i)
    return found
