DEFAULT_RESTOCK_LEVEL = 50
LOW_STOCK = 10

class Item:
    def __init__(self, name, count, price):
        self.name = name
        self.count = count
        self.price = price

    def value(self):
        return self.count * self.price

    def needs_restock(self):
        return self.count < LOW_STOCK


def restock_amount(current):
    if current >= DEFAULT_RESTOCK_LEVEL:
        return 0
    return DEFAULT_RESTOCK_LEVEL - current


def inventory_value(items):
    total = 0
    for item in items:
        total = total + item.value()
    return total


def low_stock_names(items):
    names = []
    for item in items:
        if item.needs_restock():
            names.append(item.name)
    return names


def priciest(items):
    best = None
    for item in items:
        if best is None or item.price > best.price:
            best = item
    return best


def nth_cheapest(items, n):
    ordered = sorted(items, key=lambda item: item.price)
    return ordered[n - 1]


def restock_report(items):
    lines = []
    for item in items:
        needed = restock_amount(item.count)
        if needed > 0:
            lines.append(f"{item.name}: order {needed}")
    return lines
