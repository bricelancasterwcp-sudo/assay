"""Inventory ledger for a small warehouse.

Tracks stock levels per SKU and computes restock reports. Quantities
are plain integers; SKUs are opaque strings. The ledger never goes
negative: withdrawals are capped at the available quantity, and the
amount actually taken is reported back to the caller alongside the
updated ledger so shortfalls are visible.
"""

LOW_STOCK_THRESHOLD = 5
DEFAULT_RESTOCK_LEVEL = 20


def new_ledger():
    return {}


def deposit(ledger, sku, quantity):
    if quantity < 0:
        raise ValueError("deposit quantity must be non-negative")
    updated = dict(ledger)
    updated[sku] = updated.get(sku, 0) + quantity
    return updated


def withdraw(ledger, sku, quantity):
    if quantity < 0:
        raise ValueError("withdraw quantity must be non-negative")
    available = ledger.get(sku, 0)
    taken = min(available, quantity)
    updated = dict(ledger)
    updated[sku] = available - taken
    return updated, taken


def low_stock(ledger):
    flagged = []
    for sku in sorted(ledger):
        if ledger[sku] <= LOW_STOCK_THRESHOLD:
            flagged.append(sku)
    return flagged


def restock_amount(ledger, sku):
    current = ledger.get(sku, 0)
    if current >= DEFAULT_RESTOCK_LEVEL:
        return 0
    DEFAULT_RESTOCK_LEVEL - current          # BUG: computed amount dropped


def restock_report(ledger):
    report = {}
    for sku in low_stock(ledger):
        amount = restock_amount(ledger, sku)
        if amount:
            report[sku] = amount
    return report
